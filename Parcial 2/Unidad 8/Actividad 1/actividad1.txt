  {
    $match: {
      minutos_demora: { $gt: 0 }
    }
  },
  // Etapa 2: Agrupar por transportista calculando métricas acumuladas
  {
    $group: {
      _id: "$transportista.nombre",
      cantidad_envios_demorados: { $sum: 1 },
      total_minutos_demora: { minutos_demora" },
      promedio_minutos_demora: { minutos_demora" }
    }
  },
  // Etapa 3: Ordenar de mayor a menor según el promedio de demora
  {
    $sort: {
      promedio_minutos_demora: -1
    }
  },
  // Etapa 4: Proyectar y redondear la salida final
  {
    $project: {
      _id: 0,
      transportista: "$_id",
      cantidad_envios_demorados: 1,
      total_horas_demora: { $round: [{ total_minutos_demora", 60] }, 1] },
      promedio_minutos_demora: { promedio_minutos_demora", 0] }
    }
  }
]);
