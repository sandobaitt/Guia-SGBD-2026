Base de datos origen: u8_g11
Base de datos destino (restaurada): u8_g11_restore


-------------------------------------------------------------------------------
1. COMANDO DE RESPALDO (mongodump)
-------------------------------------------------------------------------------
# Generar volcado binario BSON y metadatos JSON de la base u8_g11
mongodump --host "127.0.0.1:27017" \
  --db "u8_g11" \
  --authenticationDatabase "admin" \
  -u "adminG11" -p 'AdminPassword123*' \
  --out ~/backup_u8_g11


-------------------------------------------------------------------------------
2. VERIFICACIÓN DE ARTEFACTOS GENERADOS EN DISCO
-------------------------------------------------------------------------------
# Listar los archivos generados en la ruta del backup
ls -lh ~/backup_u8_g11/u8_g11/


-------------------------------------------------------------------------------
3. COMANDO DE RESTAURACIÓN (mongorestore)
-------------------------------------------------------------------------------
# Restaurar el volcado en la base alternativa u8_g11_restore sin alterar la original
mongorestore --host "127.0.0.1:27017" \
  --authenticationDatabase "admin" \
  -u "adminG11" -p 'AdminPassword123*' \
  --db "u8_g11_restore" \
  ~/backup_u8_g11/u8_g11/


-------------------------------------------------------------------------------
4. CONTROL DE INTEGRIDAD Y VALIDACIÓN DE CONTEOS DE DOCUMENTOS
-------------------------------------------------------------------------------
# Comando de conexión a mongosh para ejecutar las comprobaciones
mongosh "mongodb://127.0.0.1:27017/admin" -u "adminG11" -p 'AdminPassword123*'
